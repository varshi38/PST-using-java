import java.io.*;
import java.util.*;

class Result {

    public static List<Integer> maxSubarray(List<Integer> arr) {

        int maxSub = arr.get(0);
        int current = arr.get(0);
        int maxSeq = arr.get(0);

        for (int i = 1; i < arr.size(); i++) {

            current = Math.max(arr.get(i), current + arr.get(i));
            maxSub = Math.max(maxSub, current);

            if (arr.get(i) > 0) {
                maxSeq += arr.get(i);
            }
        }

        if (maxSeq < 0) {
            maxSeq = maxSub;
        }

        return Arrays.asList(maxSub, maxSeq);
    }
}

public class Solution {

    public static void main(String[] args) throws IOException {

        BufferedReader bufferedReader =
            new BufferedReader(new InputStreamReader(System.in));

        BufferedWriter bufferedWriter =
            new BufferedWriter(new FileWriter(System.getenv("OUTPUT_PATH")));

        int t = Integer.parseInt(bufferedReader.readLine().trim());

        for (int tItr = 0; tItr < t; tItr++) {

            int n = Integer.parseInt(bufferedReader.readLine().trim());

            String[] arrTemp =
                bufferedReader.readLine().trim().split(" ");

            List<Integer> arr = new ArrayList<>();

            for (int i = 0; i < n; i++) {
                arr.add(Integer.parseInt(arrTemp[i]));
            }

            List<Integer> result = Result.maxSubarray(arr);

            for (int i = 0; i < result.size(); i++) {

                bufferedWriter.write(
                    String.valueOf(result.get(i))
                );

                if (i != result.size() - 1) {
                    bufferedWriter.write(" ");
                }
            }

            bufferedWriter.newLine();
        }

        bufferedReader.close();
        bufferedWriter.close();
    }
}
